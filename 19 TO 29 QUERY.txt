SQL=>DECLARE
    rollno       NUMBER;
    name         VARCHAR2(50);
    mark1        NUMBER;
    mark2        NUMBER;
    mark3        NUMBER;
    total        NUMBER;
    percentage   NUMBER;
    class        VARCHAR2(20);
BEGIN
    rollno := &rollno;
    name := '&name';
    mark1 := &mark1;
    mark2 := &mark2;
    mark3 := &mark3;

    total := mark1 + mark2 + mark3;
    percentage := total / 3;

    IF percentage >= 60 THEN
        class := 'First Class';
    ELSIF percentage >= 50 THEN
        class := 'Second Class';
    ELSIF percentage >= 40 THEN
        class := 'Third Class';
    ELSE
        class := 'Fail';
    END IF;

    DBMS_OUTPUT.PUT_LINE('Roll No    : ' || rollno);
    DBMS_OUTPUT.PUT_LINE('Name       : ' || name);
    DBMS_OUTPUT.PUT_LINE('Total      : ' || total);
    DBMS_OUTPUT.PUT_LINE('Percentage : ' || percentage || '%');
    DBMS_OUTPUT.PUT_LINE('Class      : ' || class);
END;
/

    b NUMBER := 1;
